  <script>
    let studyData = null;
    let cardIdx = 0;
    let isFlipped = false;
    let quizIdx = 0;
    let score = 0;

    // Guardar API Key localmente para que Amanda no la escriba siempre
    window.onload = () => {
      const savedKey = localStorage.getItem('amanda_api_key');
      if (savedKey) document.getElementById('api-key').value = savedKey;
    };

    function toggleTheme() {
      const isDark = document.body.getAttribute('data-theme') === 'dark';
      document.body.setAttribute('data-theme', isDark ? 'light' : 'dark');
    }

    function switchTab(tabId, btn) {
      document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
      document.getElementById(tabId).classList.add('active');
      btn.classList.add('active');
    }

    async function generateMaterial() {
      const apiKey = document.getElementById('api-key').value.trim();
      const text = document.getElementById('study-text').value.trim();
      const count = document.getElementById('item-count').value;

      if (!apiKey || !text) {
        alert("Por favor ingresa la clave de acceso y el tema/texto para estudiar.");
        return;
      }

      localStorage.setItem('amanda_api_key', apiKey);

      document.getElementById('setup-panel').style.display = 'none';
      document.getElementById('loader').style.display = 'block';

      const prompt = `Actúa como un profesor experto en Enfermería. Analiza el siguiente contenido de clase y genera un JSON estricto con ${count} temas de recorrido clave, ${count} flashcards y ${count} preguntas de examen de opción múltiple.

FORMATO JSON ESPERADO:
{
  "recorrido": [{"tema": "...", "detalles": "...", "datoClave": "..."}],
  "flashcards": [{"pregunta": "...", "respuesta": "..."}],
  "quiz": [{"pregunta": "...", "opciones": ["...", "...", "...", "..."], "correcta": 0, "explicacion": "..."}]
}

TEXTO DE CLASE:
${text}`;

      try {
        const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{ parts: [{ text: prompt }] }],
            generationConfig: { responseMimeType: "application/json" }
          })
        });

        const data = await res.json();
        const jsonText = data.candidates[0].content.parts[0].text;
        studyData = JSON.parse(jsonText);

        renderRecorrido();
        initCards();
        initQuiz();

        document.getElementById('loader').style.display = 'none';
        document.getElementById('study-panel').style.display = 'block';
      } catch (err) {
        alert("Ocurrió un error al procesar el material. Verifica tu clave de API e inténtalo nuevamente.");
        document.getElementById('loader').style.display = 'none';
        document.getElementById('setup-panel').style.display = 'block';
      }
    }

    function renderRecorrido() {
      const container = document.getElementById('recorrido');
      container.innerHTML = studyData.recorrido.map(item => `
        <div class="card">
          <h4 style="color:var(--primary); font-size:1.1rem;">${item.tema}</h4>
          <p style="margin: 0.6rem 0; font-size:0.95rem; line-height:1.4;">${item.detalles}</p>
          <div style="background:rgba(236, 72, 153, 0.1); border-left: 4px solid var(--accent); padding:0.6rem; font-weight:600; font-size:0.9rem; border-radius:0 8px 8px 0;">
            💡 Dato Clave de Enfermería: ${item.datoClave}
          </div>
        </div>
      `).join('');
    }

    function initCards() {
      cardIdx = 0;
      isFlipped = false;
      showCard();
    }

    function showCard() {
      const card = studyData.flashcards[cardIdx];
      document.getElementById('card-text').innerText = isFlipped ? card.respuesta : card.pregunta;
      document.getElementById('card-count').innerText = `Tarjeta ${cardIdx + 1} de ${studyData.flashcards.length}`;
    }

    function flipCard() {
      isFlipped = !isFlipped;
      showCard();
    }

    function nextCard() {
      cardIdx = (cardIdx + 1) % studyData.flashcards.length;
      isFlipped = false;
      showCard();
    }

    function initQuiz() {
      quizIdx = 0;
      score = 0;
      showQuestion();
    }

    function showQuestion() {
      const q = studyData.quiz[quizIdx];
      document.getElementById('quiz-progress').innerText = `Pregunta ${quizIdx + 1} de ${studyData.quiz.length}`;
      document.getElementById('quiz-score').innerText = `Puntos: ${score}`;
      document.getElementById('quiz-question').innerText = q.pregunta;
      document.getElementById('quiz-exp').style.display = 'none';
      document.getElementById('next-quiz-btn').style.display = 'none';

      const options = document.getElementById('quiz-options');
      options.innerHTML = q.opciones.map((opt, i) => `
        <button class="option-btn" onclick="selectOption(${i})">${opt}</button>
      `).join('');
    }

    function selectOption(idx) {
      const q = studyData.quiz[quizIdx];
      const btns = document.querySelectorAll('.option-btn');

      btns.forEach((btn, i) => {
        btn.disabled = true;
        if (i === q.correcta) btn.classList.add('correct');
        else if (i === idx) btn.classList.add('incorrect');
      });

      if (idx === q.correcta) score++;

      const exp = document.getElementById('quiz-exp');
      exp.innerText = `💡 Explicación clínica: ${q.explicacion}`;
      exp.style.display = 'block';

      document.getElementById('next-quiz-btn').style.display = 'block';
    }

    function nextQuestion() {
      quizIdx++;
      if (quizIdx < studyData.quiz.length) {
        showQuestion();
      } else {
        alert(`🎉 Repaso completado, Amanda. Tu puntaje fue: ${score} / ${studyData.quiz.length}`);
      }
    }
  </script>
